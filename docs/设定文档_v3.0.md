```javascript
function updatePower(rho, dt) {
  let P = STATE.reactor.power;

  if (!STATE.reactor.critical && P < 1) {
    // 次临界，功率衰减
    P *= Math.exp(-dt * 2);
    if (P < 0.01) P = 0;
  } else {
    let tauEff = 10; // 等效时间常数（小时），简�?    if (rho > 0) {
      P *= Math.exp(rho * dt / tauEff * 50);
    } else {
      P *= Math.exp(rho * dt / tauEff * 20);
    }
  }

  // 低功率不稳定
  if (P > 0 && P < CONFIG.P_NOMINAL * 0.2) {
    let instability = (1 - P / (CONFIG.P_NOMINAL * 0.2)) * 0.1;
    P *= (1 + (Math.random() - 0.5) * instability);
  }

  // 极低功率自动停摆
  if (P < 160 && P > 0 && STATE.reactor.critical) {
    if (Math.random() < 0.1) {
      P = 0;
      STATE.reactor.critical = false;
    }
  }

  P = Math.max(0, P);
  STATE.reactor.power = P;
  STATE.reactor.powerPercent = P / CONFIG.P_NOMINAL * 100;
}
```

```javascript
function thermalStep(dt) {
  let zones = STATE.thermal.zones;
  let totalPower = STATE.reactor.power;
  let totalFlow = STATE.thermal.coolantFlow;

  // 功率分配（中心高，边缘低�?  let powerDist = [0.30, 0.25, 0.22, 0.15, 0.08];

  for (let i = 0; i < 5; i++) {
    let zone = zones[i];
    let zonePower = totalPower * powerDist[i];
    let zoneFlow = totalFlow * powerDist[i];

    let deltaT = zonePower * 3600 / (zoneFlow * CONFIG.RHO_WATER * CONFIG.CP_WATER / 1000);
    let tIn = CONFIG.SEASON_TEMP[STATE.time.season];
    zone.tIn = tIn;
    zone.tOut = tIn + deltaT;

    if (zone.tOut > CONFIG.T_SAT) {
      zone.void = Math.min(100, (zone.tOut - CONFIG.T_SAT) * 2);
    } else {
      zone.void = 0;
    }

    let rFuel = 0.5;
    let rClad = 0.2;
    zone.tClad = (zone.tIn + zone.tOut) / 2 + zonePower * rClad;
    zone.tFuel = zone.tClad + zonePower * rFuel;

    let qDNB = 1.5;
    let qActual = zonePower / 10;
    zone.dnbr = qDNB / Math.max(qActual, 0.01);
    zone.power = zonePower;
  }

  STATE.thermal.totalVoid = zones.reduce((s, z) => s + z.void, 0) / 5;
}
```

```javascript
function xenonStep(dt) {
  let xe = STATE.xenon;
  let P_norm = STATE.reactor.power / CONFIG.P_NOMINAL;

  let dI = (0.06 * P_norm - CONFIG.LAMBDA_I * xe.iodine) * dt;
  xe.iodine += dI;
  xe.iodine = Math.max(0, xe.iodine);

  let burnup = 0.8 * P_norm * xe.xenon;
  let dXe = (0.003 * P_norm + CONFIG.LAMBDA_I * xe.iodine
             - CONFIG.LAMBDA_XE * xe.xenon - burnup) * dt;
  xe.xenon += dXe;
  xe.xenon = Math.max(0, xe.xenon);

  xe.reactivity = -xe.xenon * CONFIG.BETA_EFF * 100;
```

```javascript
function eventCheck() {
  if (STATE.time.hour % 8 !== 0) return;
  if (STATE.events.active) return;

  let diff = CONFIG.DIFFICULTY[STATE.game.difficulty];
  let baseRate = 0.25 * diff.eventRate;
  let timeFactor = 1 + STATE.time.day / 100;
  let probability = baseRate * timeFactor;

    triggerRandomEvent();
  }
}

function triggerRandomEvent() {
  let available = EVENTS.filter(e => e.condition(STATE));
  if (available.length === 0) return;

  let totalWeight = available.reduce((s, e) => s + e.weight, 0);
  let roll = Math.random() * totalWeight;
  let cumulative = 0;
  for (let event of available) {
    cumulative += event.weight;
    if (roll <= cumulative) {
      STATE.events.active = event;
      UI.showModal('event', event);
      Audio.play('warning');
      break;
    }
  }
}
```
